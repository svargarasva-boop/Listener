// Логика кнопок шеринга через принудительный вызов окон JavaScript function buildShareLinks() { const encodedUrl = encodeURIComponent(siteUrl); const encodedText = encodeURIComponent(shareText); const encodedCombined = encodeURIComponent(shareText + " " + siteUrl); // 1. Telegram const tgBtn = document.getElementById('share-tg'); if (tgBtn) { tgBtn.onclick = (e) => { e.preventDefault(); window.open(`https://t.me{encodedUrl}&text=${encodedText}`, '_blank', 'noopener,noreferrer'); }; } // 2. WhatsApp const waBtn = document.getElementById('share-wa'); if (waBtn) { waBtn.onclick = (e) => { e.preventDefault(); window.open(`https://whatsapp.com{encodedCombined}`, '_blank', 'noopener,noreferrer'); }; } // 3. Viber const viberBtn = document.getElementById('share-viber'); if (viberBtn) { viberBtn.onclick = (e) => { e.preventDefault(); window.open(`viber://forward?text=${encodedCombined}`, '_blank', 'noopener,noreferrer'); }; } // 4. X (Twitter) const xBtn = document.getElementById('share-x'); if (xBtn) { xBtn.onclick = (e) => { e.preventDefault(); window.open(`https://x.com{encodedUrl}&text=${encodedText}`, '_blank', 'noopener,noreferrer'); }; } // 5. VK const vkBtn = document.getElementById('share-vk'); if (vkBtn) { vkBtn.onclick = (e) => { e.preventDefault(); window.open(`https://vk.com{encodedUrl}&title=${encodedText}`, '_blank', 'noopener,noreferrer'); }; } // 6. Instagram const instaBtn = document.getElementById('share-insta'); if (instaBtn) { instaBtn.onclick = (e) => { e.preventDefault(); navigator.clipboard.writeText(siteUrl).then(() => { alert("Ссылка скопирована! Откройте Instagram и вставьте её в Директ или Сторис."); window.open("https://instagram.com", "_blank", 'noopener,noreferrer'); }); }; } } function shareWebsite() { const isMobile = /Android|iPhone|iPad|iPod/i.test(navigator.userAgent); if (isMobile && navigator.share) { navigator.share({ title: 'Твой Слушатель', text: 'Сервис конфиденциальной поддержки. Иногда нужно просто выговориться.', url: siteUrl }).catch(() => {}); } else { const modal = document.getElementById('customShareModal'); if (modal) { buildShareLinks(); modal.style.display = 'flex'; } } } function closeShareModal() { const modal = document.getElementById('customShareModal'); if (modal) modal.style.display = 'none'; if (copyTimeoutId) { clearTimeout(copyTimeoutId); copyTimeoutId = null; resetCopyButtonState(); } } function copyShareLink() { if (navigator.clipboard && navigator.clipboard.writeText) { navigator.clipboard.writeText(siteUrl).then(() => { const btn = document.getElementById('modalCopyBtn'); if (btn) { if (copyTimeoutId) clearTimeout(copyTimeoutId); btn.textContent = 'Успешно скопировано! ✓'; btn.style.backgroundColor = '#10b981'; copyTimeoutId = setTimeout(() => { resetCopyButtonState(); copyTimeoutId = null; }, 2000); } }); } } function resetCopyButtonState() { const btn = document.getElementById('modalCopyBtn'); if (btn) { btn.textContent = '🔗 Копировать ссылку на сайт'; btn.style.backgroundColor = '#818cf8'; } } window.addEventListener('keydown', (event) => { if (event.key === 'Escape') { closeShareModal(); } });
